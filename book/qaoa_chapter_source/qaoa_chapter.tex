\section*{How to use this chapter}
The chapter is built as a staircase. Sections~1--3 are the \emph{mathematics} (graphs, cuts, probability, the Ising
form). Sections~4--5 are the \emph{quantum mechanics} (states, Pauli algebra, gates). Sections~6--9 are \emph{QAOA itself}
(state, circuit, $p = 1$ analysis, conventions). Sections~10--13 are the \emph{advanced} material and the
\emph{implementation} (depth, optimizers, the notebook, backends, noise). Section~14 collects strategies and a map from every
assessment question to the section that teaches it; Section~15 has practice problems with answers.

Each section has: the idea in words, the precise statement, a derivation, \textbf{worked examples} solved step by step,
\textbf{common traps} (the wrong answers examiners like to offer) and a short checklist. Statements are marked as
\emph{general theory}, \emph{the notebook's convention}, or \emph{illustrative assumptions}; ``previously recorded'' results are the
outputs stored in \texttt{QAOA.ipynb}.

\begin{defn}{Conventions used everywhere in this chapter}
\begin{itemize}[leftmargin=1.2em, itemsep=1pt]
\item Graph $G = (V, E)$, nodes $0, \dots, n-1$, weights $w_{ij}$ (1 if unweighted), $W = \sum_{(i,j)\in E}w_{ij}$.
\item A split is a bitstring $z = z_0z_1\dots z_{n-1}$, node 0 written first; $\bar z$ = all bits flipped; spin $s_i = (-1)^{z_i}$.
\item $C(z) = \sum_{(i,j)} w_{ij}[z_i\neq z_j]$; \ \textbf{notebook:} $H_C = \sum_{(i,j)}\frac{w_{ij}}{2}Z_iZ_j$, $C = \frac W2 - H_C$, $H_M = \sum_iX_i$.
\item $\ket{\bg,\bb} = e^{-i\beta_pH_M}e^{-i\gamma_pH_C}\cdots e^{-i\beta_1H_M}e^{-i\gamma_1H_C}\ket+^{\otimes n}$; $F_p = \braket{C}$; $r = F_p/C_{\max}$.
\item $R_X(\theta) = e^{-i\theta X/2}$, $R_Z(\theta) = e^{-i\theta Z/2}$; two-qubit basis order $\ket{00}, \ket{01}, \ket{10}, \ket{11}$.
\end{itemize}
\end{defn}

\section{Graphs, cuts and Max-Cut}
\subsection{Graphs}
A graph is a set of nodes joined by edges. The \textbf{degree} $d_i$ of a node is its number of edges. Every edge has two ends,
so
\[ \sum_{i\in V} d_i = 2|E| \qquad\text{(handshake lemma)}, \]
and therefore the number of odd-degree nodes is always even. A graph is \textbf{bipartite} if its nodes can be coloured with
two colours so that every edge joins different colours; this is possible exactly when the graph has no \textbf{odd cycle}.

\subsection{Cuts}
\begin{defn}{Cut and Max-Cut}
For a split $z$, the cut value is $C(z) = \sum_{(i,j)\in E} w_{ij}\,[z_i \neq z_j]$. Max-Cut asks for
$C_{\max} = \max_z C(z)$ and an optimal $z$.
\end{defn}
Because flipping all bits swaps the two group labels, $C(\bar z) = C(z)$: the $2^n$ bitstrings describe only $2^{n-1}$ different
partitions, and optimal bitstrings always come in complementary pairs.

\begin{keyres}{when can every edge be cut?}
$C_{\max} = |E|$ (unit weights) $\iff$ $G$ is bipartite $\iff$ $G$ has no odd cycle. Every odd cycle leaves at least one of its
edges uncut. Standard values: $C_{\max}(C_n) = n$ for even $n$ and $n - 1$ for odd $n$; $C_{\max}(K_n) = \lfloor n/2\rfloor\lceil n/2\rceil$;
$C_{\max}(K_{a,b}) = ab$; Petersen graph: $12$ of $15$.
\end{keyres}

\begin{wexample}{the house graph}
Square $0{-}1{-}2{-}3{-}0$ plus node 4 joined to 0 and 1.
\begin{steps}
\item Bound: the triangle $0, 1, 4$ is odd, so $C \le 6 - 1 = 5$.
\item Construct: $z_0z_1z_2z_3 = 0101$ cuts the 4 square edges; node 4 has one neighbour in each group, so exactly one roof edge is
cut for either value of $z_4$. $C = 5$.
\item Count: a 5-cut must cut all square edges (otherwise at most 4), so $z_0z_1z_2z_3 \in \{0101, 1010\}$ and $z_4$ is free:
4 optimal bitstrings, 2 partitions.
\end{steps}
\end{wexample}

\begin{trap}
``A graph with an odd number of nodes cannot have all edges cut.'' False: a star with 7 nodes is bipartite. What matters is an
odd \emph{cycle}. Equally, do not confuse \emph{bitstrings} (count $2^n$) with \emph{partitions} (count $2^{n-1}$).
\end{trap}

\subsection{Complexity and classical algorithms}
Max-Cut is NP-hard: no polynomial-time exact algorithm is known. \emph{Minimum} cut, in contrast, is polynomial (max-flow/min-cut,
Stoer--Wagner). The Goemans--Williamson algorithm (semidefinite relaxation plus random hyperplane rounding) guarantees
$\mathbb E[C] \ge 0.878\,C_{\max}$ in polynomial time. Single-node-flip local search stops at a \emph{local} optimum; there each node $v$
has at least half its edges cut (flipping $v$ changes the cut by $u_v - c_v \le 0$), so $C \ge |E|/2$ --- but the local optimum need
not be global. The notebook uses brute force up to 22 nodes and local search with 200 restarts above.

\section{Random splits and sampling statistics}
\begin{keyres}{random-split baseline}
If every $z_i$ is an independent fair bit, each edge is cut with probability $\frac12$, so
\[ \mathbb E[C] = \frac W2, \qquad \mathrm{Var}(C) = \sum_{e}\frac{w_e^2}{4} \;(= |E|/4 \text{ for unit weights}). \]
The variance formula uses that the edge indicators $z_i\oplus z_j$ are \emph{pairwise} independent, even for edges sharing a node.
\end{keyres}
Consequences: $C_{\max} \ge W/2$ for every graph, and random guessing already has approximation ratio $(W/2)/C_{\max}$.
For the notebook's graph: $W/2 = 15$, $\sigma = \sqrt{7.5} = 2.74$, $C_{\max} = 25$ is $3.7\sigma$ above the mean, and only
$4/2^{15} = 1.22\times10^{-4}$ of random splits are optimal.

\subsection{Random graphs}
In $G(n, q)$ each of the $\binom n2$ pairs is an edge with probability $q$: $|E| \sim \mathrm{Binomial}(\binom n2, q)$. For $n = 15$,
$q = 0.3$: mean $31.5$, standard deviation $\sqrt{105\cdot0.21} = 4.70$. The seed-42 instance has 30 edges.

\subsection{Shots}
\begin{method}{estimates from shots}
\begin{itemize}[itemsep=1pt]
\item A $\pm1$ observable such as $Z_iZ_j$ has variance $1 - \mu^2 \le 1$: standard error $\le 1/\sqrt S$. A $0/1$ indicator has variance $\le\frac14$.
\item A general observable with spread $\sigma$: standard error $\sigma/\sqrt S$.
\item With success probability $P$ per shot: number of successes in $S$ shots is Binomial($S, P$), $\sigma = \sqrt{SP(1-P)}$;
$P(\text{none in } k) = (1-P)^k$; shots for confidence $c$: $S \ge \ln(1-c)/\ln(1-P)$.
\end{itemize}
\end{method}
\begin{wexample}{the recorded $P_{\text{opt}} = 0.06$}
Expected optimal shots in 4000: $240$ (exactly what was recorded), $\sigma = \sqrt{4000\cdot0.06\cdot0.94} = 15.0$.
$P(\text{none in } 10) = 0.94^{10} = 0.539$. For $99\,\%$ confidence: $S \ge \ln0.01/\ln0.94 = 74.4$, i.e.\ $75$ shots.
\end{wexample}

\section{From bits to spins: QUBO and the cost Hamiltonian}
\subsection{Two identities}
\[ [a \neq b] = a + b - 2ab \quad(a, b \in \{0,1\}), \qquad [z_i\neq z_j] = \frac{1 - s_is_j}{2} \quad(s = (-1)^z). \]
The first gives the QUBO form, e.g.\ for the path $0{-}1{-}2$: $C = z_0 + 2z_1 + z_2 - 2z_0z_1 - 2z_1z_2$ (the linear coefficient of
$z_i$ is its weighted degree). The second gives the Ising form:
\[ C(z) = \frac W2 - \sum_{(i,j)}\frac{w_{ij}}{2}s_is_j . \]
\subsection{The cost Hamiltonian}
$Z$ has eigenvalue $s_i$ on $\ket{z_i}$, so $Z_iZ_j\ket z = s_is_j\ket z$ and
\begin{keyres}{cost Hamiltonian (notebook convention)}
\[ H_C = \sum_{(i,j)\in E}\frac{w_{ij}}{2}Z_iZ_j, \qquad C = \frac W2\,I - H_C, \qquad E_0(H_C) = \frac W2 - C_{\max}. \]
$H_C$ is diagonal; its ground states are the optimal splits. The code (\texttt{cost\_hamiltonian()}) builds exactly this, without
the constant.
\end{keyres}
\begin{wexample}{QUBO to Ising}
$f = 2x_0x_1 - x_0 - x_1$ with $x = (1 - s)/2$: $2x_0x_1 = \frac12(1 - s_0 - s_1 + s_0s_1)$, $-x_0 - x_1 = -1 + \frac12(s_0 + s_1)$; the
linear terms cancel: $f = \frac12(s_0s_1 - 1) = -[x_0\neq x_1]$.
\end{wexample}
\begin{wexample}{the triangle is frustrated}
$H_C = \frac12(Z_0Z_1 + Z_1Z_2 + Z_0Z_2)$: $+\frac32$ on $000, 111$ and $-\frac12$ on the other six strings. All three products cannot be
$-1$ since their product is $s_0^2s_1^2s_2^2 = 1$. So $E_0 = -\frac12$ (six-fold) and $C_{\max} = 2$.
\end{wexample}
\begin{trap}
Sign errors: large cuts are \emph{low} energy. $E_0 = W/2 - C_{\max}$, not $-C_{\max}$, and not $+(\dots)$.
\end{trap}

\section{Qubits, measurement and Pauli algebra}
\subsection{States and measurement}
$n$ qubits: $\ket\psi = \sum_z a_z\ket z$, $\sum|a_z|^2 = 1$; measuring returns $z$ with probability $|a_z|^2$. For a diagonal observable
$\langle H\rangle = \sum_z|a_z|^2H(z)$; per edge $\langle C_{ij}\rangle = \frac12(1 - \langle Z_iZ_j\rangle) = P(\text{edge cut})$. For a product
state $\langle Z_0Z_1\rangle = \langle Z_0\rangle\langle Z_1\rangle$ with $\langle Z\rangle = P(0) - P(1)$.

Global phases are unobservable; complex conjugation of all amplitudes leaves all probabilities unchanged. A state vector needs
$16\times2^n$ bytes in fp64 ($8\times2^n$ in fp32): 32 qubits fit on an 80 GB GPU in fp64, 33 in fp32.

\subsection{Pauli algebra}
$X^2 = Y^2 = Z^2 = I$; $XY = iZ$, $YZ = iX$, $ZX = iY$; reversing the order flips the sign. Pauli strings commute iff they
anticommute at an even number of positions: $Z_0Z_1$ commutes with $X_0X_1$ but not with $X_0$, $X_1$ or $Y_0Z_1$. Tensor products
multiply factor-wise: $(Z\otimes Z)(X\otimes X) = (ZX)\otimes(ZX) = -Y\otimes Y$.

\begin{keyres}{exponential of a Pauli}
If $P^2 = I$: $e^{-i\phi P} = \cos\phi\,I - i\sin\phi\,P$. Conjugation rotates by twice the angle:
\[ e^{i\beta X}Ze^{-i\beta X} = \cos2\beta\,Z + \sin2\beta\,Y . \]
\end{keyres}

\section{Gates: rotations and CNOT}
$R_X(\theta) = e^{-i\theta X/2}$, so $R_X(2\beta) = e^{-i\beta X}$ and $R_X(2\beta)\ket0 = \cos\beta\ket0 - i\sin\beta\ket1$ (bit flip with
probability $\sin^2\beta$). $R_Z(\theta) = \operatorname{diag}(e^{-i\theta/2}, e^{i\theta/2})$ changes only phases; after a Hadamard,
$P(0) = \cos^2(\theta/2)$ for $HR_Z(\theta)\ket+$.

\begin{keyres}{CNOT conjugation and the $ZZ$ rotation}
CNOT (control $i$, target $j$): $X_i \to X_iX_j$, $Z_j \to Z_iZ_j$, $Z_i$ and $X_j$ unchanged. Hence
\[ \mathrm{CNOT}_{ij}\,R_Z^{(j)}(\theta)\,\mathrm{CNOT}_{ij} = e^{-i\theta Z_iZ_j/2} = \operatorname{diag}(e^{-i\theta/2}, e^{i\theta/2}, e^{i\theta/2}, e^{-i\theta/2}). \]
\end{keyres}
\begin{wexample}{entangling power of the cost gate}
$e^{-i\gamma Z\otimes Z/2}\ket{++}$ has amplitudes $\frac12e^{\mp i\gamma/2}$; concurrence $2|c_{00}c_{11} - c_{01}c_{10}| = |\sin\gamma|$: maximal at
$\gamma = \pi/2$, zero at $\gamma \in \pi\mathbb Z$.
\end{wexample}

\section{The QAOA state}
\begin{defn}{depth-$p$ QAOA (notebook convention)}
\[ \ket{\bg,\bb} = \prod_{l=p}^{1} e^{-i\beta_lH_M}e^{-i\gamma_lH_C}\,\ket+^{\otimes n}, \qquad F_p = \braket{\bg,\bb|C|\bg,\bb},\qquad r = F_p/C_{\max}. \]
Layer 1 acts first. The notebook stores $\texttt{params} = [\gamma_1, \dots, \gamma_p, \beta_1, \dots, \beta_p]$ (blocked); the kernel reads
\texttt{gamma = params[layer]}, \texttt{beta = params[layers + layer]}.
\end{defn}
\subsection{The ingredients}
\begin{itemize}
\item $\ket+^{\otimes n} = H^{\otimes n}\ket0^{\otimes n}$: uniform, $\langle C\rangle = W/2$. It is the eigenstate of $H_M$ with the \emph{largest}
eigenvalue $+n$. If all $\gamma_l = 0$, the mixers only add global phases and $F_p = W/2$.
\item $e^{-i\gamma H_C}$ is diagonal: $\ket z \mapsto e^{-i\gamma H_C(z)}\ket z$. All $Z_iZ_j$ commute, so it factorises exactly over the edges.
It writes the cut value into the phase and changes no probability.
\item $e^{-i\beta H_M} = \bigotimes_i R_X(2\beta)$: no entanglement; it makes neighbouring splits interfere. $[H_M, H_C] \neq 0$, so
$e^{-i\beta H_M}e^{-i\gamma H_C} \neq e^{-i(\beta H_M + \gamma H_C)}$.
\end{itemize}
\subsection{Why it should work: the adiabatic picture}
Slow evolution keeps a system in the same eigenstate position. Trotterising a slow interpolation gives alternating layers with
special angles, so $F_p^* \to C_{\max}$ as $p\to\infty$; QAOA optimises the angles instead. Since $\ket+^{\otimes n}$ is the \emph{top} state of
$+H_M$, the matching target is the top state of the final Hamiltonian; to reach the ground state of $H_C$ interpolate to $-H_C$, i.e.\
choose $\gamma$ and $\beta$ of \textbf{opposite sign} in the notebook's convention.

\begin{wexample}{the notebook's start point}
The default ``ramp'' (\texttt{linspace(0.2, 0.6, p)} for $\gamma$, \texttt{linspace(0.6, 0.2, p)} for $\beta$) gives, for $p = 2$,
$\bg = (0.2, 0.6)$, $\bb = (0.6, 0.2)$ --- both positive. An exact calculation (NumPy and CUDA-Q agree) gives $\langle C\rangle = 9.18 < 15$:
worse than random. COBYLA recovers; the best angles found ($\bg \approx (0.367, 0.690)$, $\bb \approx (1.117, -0.275)$, and
$1.117 \equiv 1.117 - \pi/2 = -0.454$ by the $\pi/2$ period) have opposite signs.
\end{wexample}

\section{The circuit, gate counts and the code}
\begin{center}
\begin{quantikz}[row sep=0.3cm, column sep=0.3cm]
\lstick{$\ket{0}$} & \gate{H} & \ctrl{1} & \qw & \ctrl{1} & \gate{R_X(2\beta)} & \meter{}\\
\lstick{$\ket{0}$} & \gate{H} & \targ{} & \gate{R_Z(\gamma w)} & \targ{} & \gate{R_X(2\beta)} & \meter{}
\end{quantikz}
\end{center}
One edge, one layer. In general every layer has, per edge, CNOT--$R_Z(\gamma w)$--CNOT on (control $i$, target $j$) and then $R_X(2\beta)$ on
every qubit.
\begin{keyres}{counts}
gates $= n + p(3|E| + n)$; CNOTs $= 2p|E|$; $R_Z$: $p|E|$; $R_X$: $pn$; parameters $2p$. Default $(15, 30, 2)$: 225 gates, 120 CNOTs
(recorded: \texttt{h 15, cx 120, rz 60, rx 30}, depth 64).
\end{keyres}
\begin{trap}
The factor 2 in \texttt{rx(2.0 * beta)} and the absence of $\frac12$ in \texttt{rz(gamma * w)} both come from the half-angle convention
$e^{-i\theta P/2}$; they are not part of $H_M$ or $H_C$.
\end{trap}

\subsection{The objective and the optimizer in the notebook}
The objective calls \texttt{cudaq.observe(kernels["state"], H\_C, \dots).expectation()}, records $\langle C\rangle = W/2 - \langle H_C\rangle$ and returns
$\langle H_C\rangle$ to \texttt{scipy.optimize.minimize} (COBYLA, \texttt{maxiter = 80}, \texttt{rhobeg = 0.2}: the initial trust-region radius).
Minimising $\langle H_C\rangle$ is maximising $\langle C\rangle$. The final \texttt{cudaq.sample} uses the \emph{best evaluated} angles
(\texttt{argmax(history\_cut)}), not necessarily \texttt{opt.x}; every sample's cut is computed classically.

\section{Exact analysis at $p = 1$}
\begin{keyres}{one edge}
$\langle Z_0Z_1\rangle = \sin4\beta\,\sin\gamma$, \qquad $F_1 = \frac12(1 - \sin4\beta\sin\gamma)$.
Maximum 1 at $(\frac\pi2, -\frac\pi8)$; at $(\frac\pi2, \frac\pi8)$ the edge is \emph{never} cut.
\end{keyres}
\textbf{Derivation.} Mixer: $Z_0Z_1 \to (cZ_0 + sY_0)(cZ_1 + sY_1)$ with $c = \cos2\beta$, $s = \sin2\beta$. Cost: $Z_0Z_1$, $Y_0Y_1$ unchanged;
$Y_0Z_1 \to \cos\gamma\,Y_0Z_1 + \sin\gamma\,X_0$ and $Z_0Y_1 \to \cos\gamma\,Z_0Y_1 + \sin\gamma\,X_1$. In $\ket{++}$ only $X_0$, $X_1$ survive (value 1):
$\langle Z_0Z_1\rangle = 2cs\sin\gamma$.

\begin{keyres}{triangle-free graphs (Wang et al.\ 2018, translated to the notebook's sign)}
\[ \langle Z_uZ_v\rangle = \tfrac12\sin4\beta\,\sin\gamma\,\big(\cos^{d_u-1}\gamma + \cos^{d_v-1}\gamma\big). \]
Ring ($d = 2$): $\langle C_{uv}\rangle = \frac12 - \frac14\sin4\beta\sin2\gamma \le \frac34$: $r = \frac34$. 3-regular graphs: $r \ge 0.6924$ (Farhi et al.).
\end{keyres}
\begin{wexample}{the path $0{-}1{-}2$ at $(\frac\pi2, -\frac\pi8)$}
Edge $(0,1)$: $d_0 = 1$, $d_1 = 2$: $\langle Z_0Z_1\rangle = \frac12(-1)(1)(1 + 0) = -\frac12$ (note $\cos^0\gamma = 1$). Both edges alike:
$\langle C\rangle = 2\cdot\frac12(1 + \frac12) = 1.5$, $r = 0.75$. State vector: $P(010) = P(101) = 0.3125$, $P_{\text{opt}} = 0.625$.
\end{wexample}
\begin{wexample}{the triangle is solved at $p = 1$}
$H_C$ has two eigenvalues, so the cost layer sets a single relative phase between ``bad'' ($000, 111$) and ``good'' strings. At
$\gamma^* = \arctan(1/\sqrt2)$, $\beta^* = -\gamma^*/2$ one finds $F_1 = 2 = C_{\max}$ exactly (numerically verified).
\end{wexample}

\section{Symmetries and sign conventions}
\begin{itemize}
\item $\beta_l \to \beta_l + \frac\pi2$ inserts $e^{-i\frac\pi2H_M} = (-i)^nX^{\otimes n}$ (flip all bits), which commutes with $H_C$ and $H_M$ and leaves
$\ket+^{\otimes n}$ invariant: $F_p$ has period $\frac\pi2$ in each $\beta_l$.
\item $(\bg, \bb) \to (-\bg, -\bb)$ gives the complex-conjugate state (real Hamiltonians and start state): same probabilities, same $F_p$.
\item $H_C \to H_C + cI$ multiplies the state by a global phase.
\end{itemize}
\begin{trap}
Textbooks often write $e^{-i\gamma C}$. Since $C = \frac W2 - H_C$, $e^{-i\gamma C} = e^{-i\gamma W/2}e^{+i\gamma H_C}$: the notebook's circuit with
$\gamma \to -\gamma$. Literature optima such as $(\frac\pi2, \frac\pi8)$ become $(\frac\pi2, -\frac\pi8)$. The constant is harmless; the sign is not.
\end{trap}

\section{Depth, light cones and limits}
In $U^\dagger Z_uZ_vU$ every cost layer adds the neighbours of the current support; after $p$ layers only nodes within distance $p$
matter (a ring edge: $2p + 2$ qubits). Consequences: fixed-$p$ QAOA is local; for rings with $n \ge 2p + 2$ the ratio is independent of
$n$; numerically $r_p = \frac34, \frac56, \frac78$ for $p = 1, 2, 3$, following $\frac{2p+1}{2p+2}$. With optimal angles $F_{p+1}^* \ge F_p^*$
(set the extra angles to 0) and $F_p^* \to C_{\max}$; the \emph{found} value can still drop with a finite budget.

No quantum advantage for Max-Cut has been demonstrated. $p = 1$ guarantees $0.6924$ on 3-regular graphs, below
Goemans--Williamson's $0.878$, and locality limits low-depth QAOA on some graph families (Bravyi et al.\ 2020).

\section{Optimizers, gradients and noise in the objective}
\begin{itemize}
\item \textbf{Parameter shift.} For a single gate $e^{-i\theta P/2}$, $f(\theta) = a + b\cos\theta + c\sin\theta$ and
$f' = \frac12[f(\theta + \frac\pi2) - f(\theta - \frac\pi2)]$ exactly. A shared angle ($\gamma_l$ appears in $|E|$ gates) needs the chain rule:
$2|E|$ runs per $\gamma_l$, $2n$ per $\beta_l$; $2p(|E| + n) = 180$ for the default. A two-term shift of $\gamma_l$ as a whole is not exact.
\item \textbf{Finite differences} (L-BFGS-B): $2p + 1$ runs per gradient, approximate.
\item \textbf{COBYLA}: derivative-free trust region; local, no global guarantee. The recorded run stopped at its budget (80); a re-run
stopped after 75 evaluations at the trust-region lower bound, with identical results to 4 decimals.
\item \textbf{Shot noise}: standard error $\sigma_C/\sqrt S$ ($0.079$ for $\sigma_C = 2.5$, $S = 1000$). On hardware the notebook always uses shots.
\item \textbf{Better starts}: INTERP interpolates depth-$p$ optima, e.g.\ $(0.4, 0.8) \to (0.4, 0.6, 0.8)$; parameter transfer reuses angles across
graphs with similar local structure.
\end{itemize}

\section{Backends, memory, noise and variants}
\textbf{Backends} (\texttt{resolve\_backend}): \texttt{cpu}+\texttt{statevector} $\to$ \texttt{qpp-cpu}; \texttt{cpu}+\texttt{density\_matrix} $\to$
\texttt{density-matrix-cpu} ($16\times4^n$ B); \texttt{gpu}+\texttt{statevector} $\to$ \texttt{nvidia}; \texttt{gpu}+\texttt{mps} $\to$ \texttt{tensornet-mps};
\texttt{cpu}+\texttt{mps} $\to$ error (needs a GPU); \texttt{multi-gpu} needs \texttt{mpiexec}, one rank per GPU; 36 qubits in fp64 ($1.1$ TB) need 16 GPUs of 80 GB.
MPS is efficient when few edges cross each cut of the qubit order ($\chi \le 2^{p\,m_k}$): rings yes, dense graphs no.

\textbf{Noise} (illustrative global depolarising model): $\langle C\rangle_{\text{noisy}} = F\langle C\rangle + (1 - F)\frac W2$, $F = (1-\varepsilon)^{N_{\text{CNOT}}}$;
default with $\varepsilon = 0.5\,\%$: $F = 0.548$, $\langle C\rangle \approx 18.0$. Limited connectivity adds SWAPs (3 CNOTs each).

\textbf{Variants}: warm-start QAOA (initial state from a relaxation), ADAPT-QAOA (mixers from an operator pool), Quantum Alternating
Operator Ansatz (feasibility-preserving mixers such as XY), recursive QAOA (eliminate variables using $\langle Z_iZ_j\rangle$). None is in the notebook.

\section{Problem-solving strategies and question map}
\begin{method}{five habits}
\begin{enumerate}[itemsep=1pt]
\item \textbf{Convention first}: which sign ($e^{-i\gamma H_C}$ or $e^{-i\gamma C}$)? Which bit is written first? Blocked or interleaved parameters?
\item \textbf{Bound, then construct}: odd cycles bound $C_{\max}$; a 2-colouring of a large bipartite part achieves it.
\item \textbf{Count strings versus partitions}: factor 2 from $z \leftrightarrow \bar z$.
\item \textbf{Edge by edge}: $\langle C\rangle = \sum_e\frac{w_e}{2}(1 - \langle Z_uZ_v\rangle)$.
\item \textbf{Separate averages from best samples} and recorded from re-executed results.
\end{enumerate}
\end{method}

\subsection*{Summary of the chapter in ten lines}
\begin{enumerate}[itemsep=1pt]
\item $C(z) = \sum w[z_i\neq z_j]$; $C(\bar z) = C(z)$; $2^n$ strings, $2^{n-1}$ partitions.
\item Odd cycles block a perfect cut; bipartite $\iff$ $C_{\max} = |E|$; Max-Cut NP-hard, min cut easy.
\item Random split: mean $W/2$, variance $|E|/4$.
\item $[z_i\neq z_j] = \frac12(1 - s_is_j)$ $\Rightarrow$ $C = \frac W2 - H_C$, $H_C = \sum\frac w2Z_iZ_j$.
\item $e^{-i\phi P} = \cos\phi - i\sin\phi P$; conjugation rotates by $2\phi$; CNOT--$R_Z(\theta)$--CNOT $= e^{-i\theta ZZ/2}$.
\item $\ket{\bg,\bb} = \prod e^{-i\beta_lH_M}e^{-i\gamma_lH_C}\ket+^{\otimes n}$; $n + p(3|E| + n)$ gates, $2p$ parameters.
\item One edge: $F_1 = \frac12(1 - \sin4\beta\sin\gamma)$; rings $\frac34, \frac56, \frac78$; triangle exact at $p = 1$.
\item Period $\frac\pi2$ in $\beta$; $(\bg,\bb) \to -(\bg,\bb)$ symmetric; textbook $\gamma$ = $-$notebook $\gamma$.
\item Light cones make fixed-$p$ QAOA local; $F_p^* \to C_{\max}$ only as $p \to \infty$.
\item Read results honestly: $r$ vs.\ best sample vs.\ $P_{\text{opt}}$; recorded vs.\ re-executed; no speed-up claimed.
\end{enumerate}
